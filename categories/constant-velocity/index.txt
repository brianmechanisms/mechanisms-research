import os
sys.path.append(os.path.join(os.path.dirname(__file__), '..', '..', '..', 'packages', 'brianmechanisms', 'src'))
from brianmechanisms.reciprocating.linearmotion.constantvelocity._50 import mechanism50, Settings

print("=== Mechanism 50 - Constant Velocity Linear Motion (CVLM) ===")

# Create mechanism with default parameters
settings = Settings(
    R=10.0,   # Large arc radius
    H=2.0,    # Height between flat sections
    R2=0.5,   # Small arc radius (H/4)
    L=4.0,    # Length of flat sections
    R3=0.25   # Pinion radius (R2/2)
)
mech = mechanism50(settings)

print(f"\nMechanism Parameters:")
print(f"  Large arc radius (R): {mech.R}")
print(f"  Height (H): {mech.H}")
print(f"  Small arc radius (R2): {mech.R2}")
print(f"  Flat section length (L): {mech.L}")
print(f"  Pinion radius (R3): {mech.R3}")
print(f"  Total cycle distance: {mech.total_cycle_distance:.2f} units")

# Get mechanism state at different displacements
displacements = [0, 5, 10, 15, 20]
print(f"\nMechanism States:")
for d in displacements:
    state = mech.get_positions_at_displacement(d)
    print(f"  Displacement {d:.1f}: Section {state['section']} - {state['section_name']}")

# Plot and save
mech.plot(displacement=0.0).save("mechanism50_static")

# Create animation
mech.animate().save("mechanism50_animation")

print("\nMechanism 50 demonstrates constant velocity linear motion")
