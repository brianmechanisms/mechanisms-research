3:T773,// Mid-rail Clamp — sleeve + flange foot.
//
// The v-slot rail slides through the sleeve along the X axis. Two M5
// clearance holes through each side wall (4 total) drive into the
// rail's side T-channels via t-nuts. The flange below carries two
// slotted holes for bolting the assembly to a table.

rail_size  = 20;
bore_clr   = 0.4;
wall_t     = 5;
sleeve_l   = 30;
flange_w   = 60;
flange_t   = 5;
side_dia   = 5.2;
mount_dia  = 5.5;
slot_len   = 8;

bore   = rail_size + bore_clr;
outer  = bore + 2 * wall_t;

module clamp_profile() {
  difference() {
    union() {
      // Sleeve outer.
      square([outer, outer], center = true);
      // Flange hanging below.
      translate([0, -(outer / 2 + flange_t / 2)])
        square([flange_w, flange_t], center = true);
    }
    // Bore.
    square([bore, bore], center = true);
  }
}

module slot_2d(L, d) {
  hull() {
    translate([-L / 2, 0]) circle(d = d, $fn = 36);
    translate([ L / 2, 0]) circle(d = d, $fn = 36);
  }
}

module rail_mid_clamp() {
  difference() {
    // Body extruded along Y by sleeve_l, then rotated so the rail axis
    // aligns with the X direction (matching the assembly convention).
    rotate([0, 90, 0])
      linear_extrude(height = sleeve_l, center = true)
        clamp_profile();

    // Side fixing holes (4): along ±Y through the side walls.
    for (sx = [-1, 1]) for (ox = [-7.5, 7.5])
      translate([ox, sx * (bore / 2 + wall_t / 2 + 1), 0])
        rotate([90, 0, 0])
          cylinder(h = wall_t + 4, d = side_dia, $fn = 32, center = true);

    // Floor mount slots (2): vertical, along Z through the flange.
    flange_mid_z = -(outer / 2 + flange_t / 2);
    for (sy = [-1, 1])
      translate([0, sy * (flange_w / 2 - 8), flange_mid_z])
        rotate([0, 0, 0])
          linear_extrude(height = flange_t + 1, center = true)
            slot_2d(slot_len, mount_dia);
  }
}

rail_mid_clamp();
