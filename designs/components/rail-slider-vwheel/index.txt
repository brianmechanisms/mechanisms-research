3:T757,// V-Wheel Slider Carriage — plate + 4 wheels + follower pin +
// upper/lower link-pivot flanges.
//
// Plate sits on the rail's +Y face. V-wheels engage the rail. The
// follower pin sticks +Y into the cam plate. Top and bottom flanges
// extend outboard with two link-pivot holes each (spaced by par_w
// along the rail), where the parallelogram + single-bar links attach.

plate_l    = 60;
plate_h    = 30;
plate_t    = 5;
wheel_dia  = 24;
wheel_t    = 11;
wheel_sx   = 40;
wheel_sz   = 16;
pin_dia    = 8;
pin_len    = 12;
par_w      = 16;     // mm (worldScale * unit-par_w)
flange_y   = 18;
flange_t   = 4;
pivot_dia  = 5;
pivot_len  = 8;
rail_half  = 10;
clearance  = 1;

plate_in_y  = rail_half + clearance + wheel_t;
plate_y     = plate_in_y + plate_t / 2;
plate_out_y = plate_in_y + plate_t;
wheel_y     = rail_half + clearance + wheel_t / 2;

module slider() {
  // Plate
  translate([0, plate_y, 0])
    cube([plate_l, plate_t, plate_h], center = true);

  // V-wheels
  for (sx = [-1, 1]) for (sz = [-1, 1])
    translate([sx * wheel_sx / 2, wheel_y, sz * wheel_sz / 2])
      rotate([90, 0, 0])
        cylinder(h = wheel_t, d = wheel_dia, center = true, $fn = 48);

  // Follower pin
  translate([0, plate_out_y + pin_len / 2, 0])
    rotate([90, 0, 0])
      cylinder(h = pin_len, d = pin_dia, center = true, $fn = 32);

  // Upper + lower flanges with link pivots
  flange_len = max(plate_l, par_w + 12);
  for (sz = [-1, 1]) {
    flange_z = sz * (plate_h / 2 + flange_t / 2);
    translate([0, plate_out_y + flange_y / 2, flange_z])
      cube([flange_len, flange_y, flange_t], center = true);
    // Pivots
    pivot_y = plate_out_y + flange_y - 4;
    for (px = [-par_w / 2, par_w / 2])
      translate([px, pivot_y, flange_z + sz * (flange_t / 2 + pivot_len / 2)])
        cylinder(h = pivot_len, d = pivot_dia, center = true, $fn = 24);
  }
}
