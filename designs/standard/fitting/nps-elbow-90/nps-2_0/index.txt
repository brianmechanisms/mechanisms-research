3:Tb39,// 2" NPS 90° short-radius elbow — ANSI/ASME B16.3 (Class 150 malleable iron threaded fittings).
//
// Construction: a single curved tube swept along a 90° arc in the XY
// plane, with raised bands at each open face. Built via rotate_extrude
// around the Z axis sweeping the profile from t=0° to t=90°.
//
//   Profile circle centered at (Rc, 0) in the XZ-plane, radius endR.
//   Sweep CCW around Z by 90°: profile centers trace
//     (Rc·cos t, Rc·sin t, 0) for t in [0, 90°].
//
// Endpoints of the swept tube:
//   t=0  →  centerline at (Rc, 0, 0)  with the rotate_extrude's natural
//           flat cap in the XZ plane — face normal = +Y.
//           So this is "leg-B end": a pipe attached here runs along +Y,
//           with its end face flush at y = 0.
//   t=90 →  centerline at (0, Rc, 0)  with cap in the YZ plane —
//           face normal = -X. This is "leg-A end": pipe runs along -X
//           with its end face flush at x = 0. To match the convention
//           that BOTH legs extend in the +X / +Y directions away from
//           the elbow, we instead say leg-A's open face is at x = 0
//           and the attached pipe extends along -X. (Mirror this whole
//           model with mirror([1,1,0]) to flip both legs to opposite
//           directions if needed.)
//
// With Rc = A, the leg-B face is at y = 0 (perpendicular to +Y) at
// position (A, 0, 0), and the leg-A face is at x = 0 (perpendicular to
// -X) at position (0, A, 0). The "centerline intersection" — where the
// two legs' straight centerlines would meet — is at (A, A, 0) (the
// bend center). centerToEnd = A in both cases. ✓
$fa = 1; $fs = 0.4;

module bend_outer() {
  rotate_extrude(angle = 90, $fn = 96)
    translate([57, 0]) circle(r = 38, $fn = 48);
}
module bend_bore() {
  rotate_extrude(angle = 90, $fn = 96)
    translate([57, 0]) circle(r = 26.25, $fn = 48);
}

// Raised end-band rings at each open face. Band sits flush against the
// open face on the leg's inside, so it appears as a thicker collar.
// Leg-B face is the XZ plane (y = 0) at x = Rc; band is a disc in that
// plane, axis = +Y, width bw extending inward (+Y).
module band_B() {
  translate([57, 0, 0])
    rotate([-90, 0, 0])
      cylinder(h = 8, r = 50, $fn = 48);
}
// Leg-A face is the YZ plane (x = 0) at y = Rc; band axis = +X.
module band_A() {
  translate([0, 57, 0])
    rotate([0, 90, 0])
      cylinder(h = 8, r = 50, $fn = 48);
}

module std_elbow_nps_2_0() {
  difference() {
    union() {
      bend_outer();
      band_A();
      band_B();
    }
    bend_bore();
    // Drill the bands too so the through-bore is continuous.
    translate([57, -0.5, 0])
      rotate([-90, 0, 0])
        cylinder(h = 9, r = 26.25, $fn = 48);
    translate([-0.5, 57, 0])
      rotate([0, 90, 0])
        cylinder(h = 9, r = 26.25, $fn = 48);
  }
}
