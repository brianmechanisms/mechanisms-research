# Verify constant rotation speed across all phases
def verify_timing():
    settings = mechanism42.Settings(R1=90, r2=30, tangent_angle=70, num_racks=1)
    mech = mechanism42(settings)
    
    print("\nTiming Verification:")
    print(f"Phase 1 duration ratio: {mech.phase1_duration_ratio:.4f}")
    print(f"Phase 2 duration ratio: {mech.phase2_duration_ratio:.4f}")  
    print(f"Phase 3 duration ratio: {mech.phase3_duration_ratio:.4f}")
    print(f"Phase 4 duration ratio: {mech.phase4_duration_ratio:.4f}")
    print(f"Phase 5 duration ratio: {mech.phase5_duration_ratio:.4f}")
    
    # Calculate rotation speed for each phase
    total_frames = sum([mech.phase1_frames, mech.phase2_frames, mech.phase3_frames, 
                       mech.phase4_frames, mech.phase5_frames])
    
    for i, (rotation, frames) in enumerate([
        (mech.phase1_rotation, mech.phase1_frames),
        (mech.phase2_rotation, mech.phase2_frames), 
        (mech.phase3_rotation, mech.phase3_frames),
        (mech.phase4_rotation, mech.phase4_frames),
        (mech.phase5_rotation, mech.phase5_frames)
    ]):
        speed = rotation / frames * total_frames
        print(f"Phase {i+1} rotation speed: {speed:.4f} rad/time_unit")

verify_timing()
